function timeLyOpen(gubun, lineID, busNum){
	
	if(gubun=="start_end")

		$("#timetable1").attr("style", "display:black;");
		fnXHRSchOption(lineID, 'start_end_time');

		$("#timetable2").attr("style", "display:black;");
		fnXHRSchOption(lineID, 'bustimetable');
	}
}
